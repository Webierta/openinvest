import 'dart:convert';

import 'package:http/http.dart' as http;

/// Resultado de la resolución de un ISIN.
class IsinResult {
  final String isin;
  final String source;
  final String? officialName;
  final int? cnmvRegistration;
  final String? cnmvNif;

  const IsinResult({
    required this.isin,
    required this.source,
    this.officialName,
    this.cnmvRegistration,
    this.cnmvNif,
  });

  @override
  String toString() {
    return 'IsinResult('
        'isin: $isin, '
        'source: $source, '
        'officialName: $officialName, '
        'cnmvRegistration: $cnmvRegistration, '
        'cnmvNif: $cnmvNif'
        ')';
  }
}

/// Resuelve el ISIN de una IIC a partir de ticker + nombre.
///
/// Actualmente implementa:
///   SLxxx.MC
///
/// utilizando el registro oficial de la CNMV.
class IsinResolver {
  static const String _cnmvListBaseUrl =
      'https://www.cnmv.es/portal/consultas/mostrarlistados'
      '?id=5&lang=es&page=';

  static const String _cnmvSocietyUrl =
      'https://www.cnmv.es/portal/consultas/iic/sociedadiic?nif=';

  final http.Client _client;
  Map<int, _CnmvEntity>? _cnmvEntities;

  IsinResolver({http.Client? client}) : _client = client ?? http.Client();

  Future<IsinResult?> resolve({
    required String ticker,
    required String fundName,
  }) async {
    final normalizedTicker = ticker.trim().toUpperCase();

    print('');
    print('------------------------------------------------------------');
    print('IsinResolver');
    print('Ticker: $normalizedTicker');
    print('Nombre: $fundName');
    print('------------------------------------------------------------');

    final silMatch = RegExp(
      r'^SL(\d+)\.MC$',
      caseSensitive: false,
    ).firstMatch(normalizedTicker);

    if (silMatch != null) {
      final registrationNumber = int.tryParse(silMatch.group(1)!);

      if (registrationNumber == null) {
        print('ERROR: no se pudo obtener el registro CNMV');
        return null;
      }

      print(
        'Tipo detectado: SIL española '
        '(registro CNMV $registrationNumber)',
      );

      return _resolveSil(
        registrationNumber: registrationNumber,
        fundName: fundName,
      );
    }

    print('No se reconoce como ticker SIL: $normalizedTicker');
    return null;
  }

  Future<IsinResult?> _resolveSil({
    required int registrationNumber,
    required String fundName,
  }) async {
    print('');
    print('Buscando registro CNMV $registrationNumber...');

    final entities = await _loadCnmvSilIndex();

    print('Entidades CNMV cargadas: ${entities.length}');

    final entity = entities[registrationNumber];

    if (entity == null) {
      print('NO ENCONTRADO: registro CNMV $registrationNumber');
      return null;
    }

    print('');
    print('Entidad CNMV encontrada:');
    print('  Registro: ${entity.registrationNumber}');
    print('  Nombre:   ${entity.name}');
    print('  NIF:      ${entity.nif}');
    print('  URL:      ${entity.url}');

    final similarity = _nameSimilarity(fundName, entity.name);

    print('');
    print('Comparación de nombres:');
    print('  Entrada:  $fundName');
    print('  CNMV:     ${entity.name}');
    print(
      '  Similaridad: '
      '${(similarity * 100).toStringAsFixed(1)}%',
    );

    if (similarity < 0.65) {
      print('ERROR: los nombres no coinciden suficientemente.');
      return null;
    }

    print('');
    print('Consultando ficha CNMV...');
    print('NIF: ${entity.nif}');

    final html = await _getCnmvSociety(entity.nif);

    if (html == null) {
      print('ERROR: no se pudo obtener la ficha CNMV.');
      return null;
    }

    print('Ficha CNMV recibida: ${html.length} caracteres');

    final isin = _extractIsin(html);

    if (isin == null) {
      print('ERROR: no se encontró ningún ISIN.');
      return null;
    }

    print('');
    print('ISIN encontrado: $isin');
    print('Fuente: CNMV');

    return IsinResult(
      isin: isin,
      source: 'CNMV',
      officialName: entity.name,
      cnmvRegistration: registrationNumber,
      cnmvNif: entity.nif,
    );
  }

  Future<Map<int, _CnmvEntity>> _loadCnmvSilIndex() async {
    if (_cnmvEntities != null) {
      print('Usando cache del listado CNMV.');
      return _cnmvEntities!;
    }

    print('');
    print('Descargando listado CNMV de SIL...');

    final result = <int, _CnmvEntity>{};

    const maxPages = 50;

    for (var page = 1; page <= maxPages; page++) {
      print('  Consultando página CNMV $page...');

      final uri = Uri.parse('$_cnmvListBaseUrl$page');

      final response = await _client.get(
        uri,
        headers: const {'Accept': 'text/html', 'User-Agent': 'OpenInvest/1.0'},
      );

      print(
        '    HTTP ${response.statusCode} '
        '(${response.bodyBytes.length} bytes)',
      );

      if (response.statusCode != 200) {
        throw Exception(
          'CNMV respondió HTTP ${response.statusCode} '
          'al consultar la página $page',
        );
      }

      final html = utf8.decode(response.bodyBytes, allowMalformed: true);

      final pageEntities = _parseCnmvListPage(html);

      print('    Entidades encontradas: ${pageEntities.length}');

      for (final entity in pageEntities) {
        result[entity.registrationNumber] = entity;
      }

      if (pageEntities.isEmpty) {
        print('    Página sin entidades. Fin del listado.');
        break;
      }

      if (!_hasNextPage(html)) {
        print('    No se detecta página siguiente.');
        break;
      }
    }

    print('');
    print('TOTAL entidades CNMV indexadas: ${result.length}');

    for (final number in [20, 21, 24]) {
      final entity = result[number];

      if (entity != null) {
        print('  [$number] ${entity.name} (${entity.nif})');
      } else {
        print('  [$number] NO ENCONTRADO');
      }
    }

    _cnmvEntities = result;
    return result;
  }

  List<_CnmvEntity> _parseCnmvListPage(String html) {
    final result = <_CnmvEntity>[];

    // Primero localizamos todos los enlaces <a>...</a>.
    // Después filtramos los que apuntan a sociedadiic.
    final anchorPattern = RegExp(
      r'<a\b[^>]*>.*?</a>',
      caseSensitive: false,
      dotAll: true,
    );

    for (final anchorMatch in anchorPattern.allMatches(html)) {
      final anchor = anchorMatch.group(0);

      if (anchor == null) {
        continue;
      }

      // Solo nos interesan los enlaces a fichas de sociedades de inversión.
      if (!anchor.toLowerCase().contains('sociedadiic')) {
        continue;
      }

      // Extraer href.
      final hrefMatch = RegExp(
        r'''href\s*=\s*["']([^"']+)["']''',
        caseSensitive: false,
      ).firstMatch(anchor);

      if (hrefMatch == null) {
        continue;
      }

      final href = hrefMatch.group(1);

      if (href == null || href.isEmpty) {
        continue;
      }

      // Extraer NIF del parámetro de la URL.
      final nifMatch = RegExp(
        r'''[?&]nif=([^&"' ]+)''',
        caseSensitive: false,
      ).firstMatch(href);

      if (nifMatch == null) {
        continue;
      }

      final nif = nifMatch.group(1);

      if (nif == null || nif.isEmpty) {
        continue;
      }

      // Extraer el texto visible del enlace.
      final rawName = anchor
          .replaceFirst(
            RegExp(r'^.*?>', caseSensitive: false, dotAll: true),
            '',
          )
          .replaceFirst(
            RegExp(r'</a>\s*$', caseSensitive: false, dotAll: true),
            '',
          );

      final name = _cleanHtmlText(rawName);

      if (name.isEmpty) {
        continue;
      }

      // Buscamos el número de registro cerca del enlace.
      final start = anchorMatch.start;
      final end = (anchorMatch.end + 1500 < html.length)
          ? anchorMatch.end + 1500
          : html.length;

      final surroundingHtml = html.substring(start, end);

      final registration = _extractRegistrationNumber(surroundingHtml);

      if (registration == null) {
        continue;
      }

      result.add(
        _CnmvEntity(
          registrationNumber: registration,
          name: name,
          nif: nif,
          url: _absoluteCnmvUrl(href),
        ),
      );
    }

    return result;
  }

  int? _extractRegistrationNumber(String html) {
    final patterns = [
      RegExp(
        r'(?:Número y fecha de registro oficial|'
        r'Numero y fecha de registro oficial|'
        r'Official registration number and date)'
        r'[\s\S]{0,300}?'
        r'(?:>|:|\s)(\d+)\s*[-/]',
        caseSensitive: false,
      ),
      RegExp(
        r'(?:registro oficial|registration number)'
        r'[\s\S]{0,300}?(\d+)',
        caseSensitive: false,
      ),
    ];

    for (final pattern in patterns) {
      final match = pattern.firstMatch(html);

      if (match != null) {
        final value = int.tryParse(match.group(1)!);
        if (value != null) {
          return value;
        }
      }
    }

    return null;
  }

  Future<String?> _getCnmvSociety(String nif) async {
    final uri = Uri.parse('$_cnmvSocietyUrl$nif');

    final response = await _client.get(
      uri,
      headers: const {'Accept': 'text/html', 'User-Agent': 'OpenInvest/1.0'},
    );

    if (response.statusCode != 200) {
      print('CNMV ficha HTTP ${response.statusCode}');
      return null;
    }

    return utf8.decode(response.bodyBytes, allowMalformed: true);
  }

  String? _extractIsin(String html) {
    final nearIsin = RegExp(
      r'ISIN[\s\S]{0,500}?([A-Z]{2}[A-Z0-9]{9}\d)',
      caseSensitive: false,
    ).firstMatch(html);

    if (nearIsin != null) {
      return nearIsin.group(1)?.toUpperCase();
    }

    final generic = RegExp(
      r'\b[A-Z]{2}[A-Z0-9]{9}\d\b',
      caseSensitive: false,
    ).firstMatch(html);

    return generic?.group(0)?.toUpperCase();
  }

  double _nameSimilarity(String a, String b) {
    final na = _normalizeName(a);
    final nb = _normalizeName(b);

    if (na.isEmpty || nb.isEmpty) {
      return 0;
    }

    if (na == nb) {
      return 1.0;
    }

    if (na.contains(nb) || nb.contains(na)) {
      return 0.90;
    }

    final tokensA = na.split(' ').toSet();
    final tokensB = nb.split(' ').toSet();

    final intersection = tokensA.intersection(tokensB).length;
    final union = tokensA.union(tokensB).length;

    if (union == 0) {
      return 0;
    }

    return intersection / union;
  }

  String _normalizeName(String value) {
    var result = value.toUpperCase();

    const replacements = {
      'Á': 'A',
      'À': 'A',
      'Ä': 'A',
      'Â': 'A',
      'É': 'E',
      'È': 'E',
      'Ë': 'E',
      'Ê': 'E',
      'Í': 'I',
      'Ì': 'I',
      'Ï': 'I',
      'Î': 'I',
      'Ó': 'O',
      'Ò': 'O',
      'Ö': 'O',
      'Ô': 'O',
      'Ú': 'U',
      'Ù': 'U',
      'Ü': 'U',
      'Û': 'U',
      'Ñ': 'N',
    };

    replacements.forEach((from, to) {
      result = result.replaceAll(from, to);
    });

    result = result
        .replaceAll(RegExp(r'[^A-Z0-9]+'), ' ')
        .replaceAll(RegExp(r'\s+'), ' ')
        .trim();

    result = result
        .replaceAll(' SOCIEDAD ANONIMA', '')
        .replaceAll(' SIL', '')
        .replaceAll(' S A', '')
        .trim();

    return result;
  }

  String _cleanHtmlText(String value) {
    return value
        .replaceAll(RegExp(r'<[^>]+>'), ' ')
        .replaceAll('&nbsp;', ' ')
        .replaceAll('&amp;', '&')
        .replaceAll('&quot;', '"')
        .replaceAll('&#39;', "'")
        .replaceAll(RegExp(r'\s+'), ' ')
        .trim();
  }

  String _absoluteCnmvUrl(String href) {
    if (href.startsWith('http://') || href.startsWith('https://')) {
      return href;
    }

    if (href.startsWith('/')) {
      return 'https://www.cnmv.es$href';
    }

    return 'https://www.cnmv.es/$href';
  }

  bool _hasNextPage(String html) {
    return RegExp(
      r'>\s*(?:»|Siguiente|Next)\s*<',
      caseSensitive: false,
    ).hasMatch(html);
  }

  void dispose() {
    _client.close();
  }
}

class _CnmvEntity {
  final int registrationNumber;
  final String name;
  final String nif;
  final String url;

  const _CnmvEntity({
    required this.registrationNumber,
    required this.name,
    required this.nif,
    required this.url,
  });

  @override
  String toString() {
    return '_CnmvEntity('
        'registrationNumber: $registrationNumber, '
        'name: $name, '
        'nif: $nif, '
        'url: $url'
        ')';
  }
}
